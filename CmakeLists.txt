cmake_minimum_required(VERSION 3.16)
project(DXGIDesktopCapture)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# 添加可执行文件
add_executable(DXGIDesktopCapture
    src/main.cpp
    src/DXGICapture.cpp
    src/MouseHandler.cpp
)

# 链接库
target_link_libraries(DXGIDesktopCapture
    d3d11
    dxgi
    d3dcompiler
)

# 设置编译选项
if(MSVC)
    target_compile_options(DXGIDesktopCapture PRIVATE /W4)
else()
    target_compile_options(DXGIDesktopCapture PRIVATE -Wall -Wextra -Wpedantic)
endif()
